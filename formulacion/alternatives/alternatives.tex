\section{Configuración y Evaluación Preliminar de Alternativas de Solución}

\subsection{Configuración de las Alternativas de Solución}


\textbf{Alternativa A1: Digitalización Prioritaria}
\begin{itemize}
\item \textbf{Incluye:} Procedimientos automáticos, Presencia de software especializado, Aumento de la infraestructura tecnológica, Alta productividad de los funcionarios (capacitación)
\item \textbf{Excluye:} Alta integración locativa completa (solo adecuaciones mínimas TI)
\item \textbf{Justificación:} Máxima automatización de trámites sin grandes obras físicas
\end{itemize}

\textbf{Alternativa A2: Mejora Locativa y Humana}
\begin{itemize}
\item \textbf{Incluye:} Alta integración locativa (ubicaciones adecuadas, estudios tráfico), Suficiente personal de apoyo, Alta productividad de los funcionarios
\item \textbf{Excluye:} Desarrollo software complejo
\item \textbf{Justificación:} Optimiza lo existente sin grandes inversiones TI
\end{itemize}

\textbf{Alternativa A3: Solución Híbrida}
\begin{itemize}
\item \textbf{Incluye:} Alta integración locativa básica, Procedimientos automáticos, Suficiente personal de apoyo, Infraestructura tecnológica media
\item \textbf{Excluye:} Migración completa a nube (solo servidores locales mejorados)
\item \textbf{Justificación:} Equilibrio costo-beneficio con impacto inmediato
\end{itemize}

\textbf{Alternativa A4: Solución Integral Completa (Transformacional)}
\begin{itemize}
\item \textbf{Incluye:} TODOS los componentes del árbol de soluciones
\item \textbf{Excluye:} Ninguna (ejecuta todo secuencialmente)
\item \textbf{Justificación:} Máxima ambición, alineada con Gobierno Digital
\end{itemize}

\subsection{Análisis Detallado por Criterio}

\subsubsection{Viabilidad Técnica}
\begin{itemize}
\item \textbf{A1:} \textcolor{orange}{Media-Alta} - Requiere desarrollo software desde requerimientos (RF propuestos), pero usa tecnologías estándar (APIs RUNT, PSE). Riesgo moderado, metodología ágil mitiga.
\item \textbf{A2:} \textcolor{green}{Alta} - Obras locativas estándar (experiencia municipal). Proveedores externos disponibles. Riesgo bajo.
\item \textbf{A3:} \textcolor{green}{Alta} - Combina obras probadas + desarrollo acotado. Integración sencilla (servidores locales).
\item \textbf{A4:} \textcolor{orange}{Media} - Complejidad alta (desarrollo full + obras + integraciones). Requiere equipo TI externo fortalecido.
\end{itemize}

\subsubsection{Aceptabilidad por la Comunidad}
\begin{itemize}
\item \textbf{A1/A3/A4:} Alta - Ciudadanos priorizan trámites rápidos sobre infraestructura física.
\item \textbf{A2:} Alta - Mejora filas y accesos, pero sin digitalización el impacto es menor.
\end{itemize}

\subsubsection{Financiamiento Requerido vs Disponible}
\begin{itemize}
\item \textbf{A1:} Media (\$450M) - Software + equipos + capacitación. Dentro presupuesto TI Secretaría.
\item \textbf{A2:} Baja (\$750M) - Obras locativas + mobiliario. Excede presupuesto municipal corriente.
\item \textbf{A3:} Alta (\$600M) - Balanceado, justificado por ROI rápido (18 meses).
\item \textbf{A4:} Baja (\$1.200M) - Requiere regalías o Fonade, inviable corto plazo.
\end{itemize}

\subsubsection{Capacidad Institucional para Ejecutar}
\begin{itemize}
\item \textbf{A1/A3:} Alta - Equipo TI existente + contratación temporal desarrolladores.
\item \textbf{A2:} Alta - Experiencia en obras públicas municipales.
\item \textbf{A4:} Media - Requiere PM profesional externo y área TI fortalecida.
\end{itemize}

\subsubsection{Impacto Ambiental}
\begin{itemize}
\item \textbf{A1/A3/A4:} Alta (positivo) - Reducción papel (digitalización), menor desplazamientos por reprocesos.
\item \textbf{A2:} Media - Obras generan residuos construcción.
\end{itemize}

\subsection{Alternativas Descartadas y Recomendada}

\textbf{Descartadas:}
\begin{itemize}
\item \textbf{A2:} Inviable financieramente. No resuelve reprocesos ni automatización (corazón del problema).
\item \textbf{A4:} Excede capacidad institucional y presupuesto disponible.
\end{itemize}

\textbf{Recomendada: A3 - Solución Híbrida}
\begin{itemize}
\item Cubre 80\% del impacto del árbol de soluciones con viabilidad alta en todos los criterios.
\item Costo manejable (\$600M), ROI demostrable (tiempo atención 120→30 min).
\item Escalable: piloto exitoso justifica A4 futura.
\end{itemize}
