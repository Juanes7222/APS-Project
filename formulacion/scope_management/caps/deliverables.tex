\subsection{Entregables del Proyecto}
\label{sec:Entregables del producto}
A continuación, se definen los entregables principales del proyecto. 

\subsubsection{Entregables de Gestión e Iniciación}

\paragraph{Acta de constitución del proyecto}
\begin{itemize}
    \item \textbf{ID EDT:} 1.1
    \item \textbf{Descripción:} Documento que formaliza el inicio del proyecto, autoriza el uso de recursos y define objetivos, alcance preliminar y principales interesados.
    \item \textbf{Criterios de aceptación:} Firma del patrocinador confirmando la disponibilidad de fondos y recursos.
    \item \textbf{Evidencia de verificación:} Documento firmado física o digitalmente.
\end{itemize}

\paragraph{Informe de diagnóstico y viabilidad}
\begin{itemize}
    \item \textbf{ID EDT:} 1.2
    \item \textbf{Descripción:} Estudio del estado actual de los trámites, tráfico de usuarios y viabilidad técnica/operativa para la implementación del sistema.
    \item \textbf{Criterios de aceptación:} Aprobado por el comité directivo y validado técnicamente.
    \item \textbf{Evidencia de verificación:} Informe formal aprobado con visto bueno del cliente.
\end{itemize}

\subsubsection{Entregables de Requisitos y Diseño}

\paragraph{Especificación de Requerimientos del Sistema (ERS)}
\begin{itemize}
    \item \textbf{ID EDT:} 2.1
    \item \textbf{Descripción:} Documentación exhaustiva de requisitos funcionales y no funcionales (gestión de turnos, integraciones, pagos, etc.) que delimita el alcance técnico.
    \item \textbf{Criterios de aceptación:} Cobertura al 100\% de las necesidades levantadas y validación formal de los requerimientos.
    \item \textbf{Evidencia de verificación:} Documento ERS avalado y firmado por los stakeholders clave.
\end{itemize}

\paragraph{Documento de Diseño de Arquitectura y Modelo de Datos (Base de Datos)}
\begin{itemize}
    \item \textbf{ID EDT:} 2.2
    \item \textbf{Descripción:} Diagramas de arquitectura en la nube, flujos de integración y el modelo de entidad-relación de la base de datos que soportará la operación a nivel nacional.
    \item \textbf{Criterios de aceptación:} Aprobación técnica que asegure escalabilidad, seguridad y eficiencia.
    \item \textbf{Evidencia de verificación:} Planos de arquitectura y esquemas de BD aprobados por el líder de TI.
\end{itemize}

\subsubsection{Entregables de Desarrollo, Calidad y Producto}

\paragraph{Módulos y Funcionalidades Específicas del Software}
\begin{itemize}
    \item \textbf{ID EDT:} 3.1
    \item \textbf{Descripción:} Código fuente y componentes desarrollados para la gestión de citas, verificación documental, conciliación de pagos, RUNT, biometría y excepciones.
    \item \textbf{Criterios de aceptación:} Código revisado (Code Review) sin vulnerabilidades graves y desplegado en el ambiente de pruebas.
    \item \textbf{Evidencia de verificación:} Repositorio de código actualizado y ambiente de staging funcional.
\end{itemize}

\paragraph{Resultados de Pruebas del Sistema (UAT y Estrés)}
\begin{itemize}
    \item \textbf{ID EDT:} 3.2
    \item \textbf{Descripción:} Informes y logs resultantes de las pruebas funcionales, de seguridad, estrés y la aceptación de los usuarios finales (User Acceptance Testing).
    \item \textbf{Criterios de aceptación:} 100\% de bugs críticos resueltos y concurrencia soportada satisfactoriamente.
    \item \textbf{Evidencia de verificación:} Matriz de trazabilidad de pruebas cubierta y actas UAT firmadas por usuarios.
\end{itemize}

\paragraph{Producto de Software Funcional (Sistema en Producción)}
\begin{itemize}
    \item \textbf{ID EDT:} 4.1
    \item \textbf{Descripción:} El Sistema Nacional Integral de Trámites completamente desplegado en infraestructura de producción (nube), interconectado y listo para operar.
    \item \textbf{Criterios de aceptación:} Puesta en vivo funcional con cero caídas en las primeras 48 horas de monitoreo intensivo.
    \item \textbf{Evidencia de verificación:} URL productiva operativa y certificados de calidad de despliegue.
\end{itemize}

\subsubsection{Entregables de Transición y Cierre}

\paragraph{Manual Técnico y Manual de Usuario}
\begin{itemize}
    \item \textbf{ID EDT:} 4.2
    \item \textbf{Descripción:} Documentación orientada a administradores IT (Manual Técnico) y a operadores/funcionarios de atención en sedes (Manual de Usuario).
    \item \textbf{Criterios de aceptación:} Redacción clara, paso a paso de los procesos y guías de soporte validadas.
    \item \textbf{Evidencia de verificación:} Documentos PDF publicados y compartidos con mesa de ayuda.
\end{itemize}

\paragraph{Registros de Capacitación}
\begin{itemize}
    \item \textbf{ID EDT:} 4.3
    \item \textbf{Descripción:} Constancias de la formación impartida a los funcionarios sobre el nuevo sistema y manejo de casos.
    \item \textbf{Criterios de aceptación:} Asistencia igual o superior al 90\% del personal objetivo.
    \item \textbf{Evidencia de verificación:} Listas de asistencia e informe de evaluación de capacitadores.
\end{itemize}

\paragraph{Acta de Cierre y Registro de Aceptación Final}
\begin{itemize}
    \item \textbf{ID EDT:} 5.1
    \item \textbf{Descripción:} Documento oficial que certifica que todos los entregables previos cumplen con la línea base del alcance y transfiere la propiedad del producto final.
    \item \textbf{Criterios de aceptación:} Validación final de que el alcance se cumplió en tiempo, costo e integridad funcional.
    \item \textbf{Evidencia de verificación:} Acta de aceptación final firmada por Patrocinador y Project Manager.
\end{itemize}
